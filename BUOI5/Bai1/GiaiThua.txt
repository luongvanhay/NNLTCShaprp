using System;

namespace NMLT.Buoi05
{
    class GiaiThua
    {
        public static void Main(string[] args)
        {
            // Khai báo biến
            int n, i;
            long giaiThua = 1;

            // Nhập dữ liệu
            Console.Write("Moi ban nhap so n: ");
            n = int.Parse(Console.ReadLine());

            // Tính giai thừa
            for (i = 1; i <= n; i++)
            {
                giaiThua = giaiThua * i;
            }

            // Xuất kết quả
            Console.Write("{0}! = ", n);
            for (i = 1; i <= n; i++)
            {
                Console.Write("{0}", i);
                if (i < n)
                    Console.Write(".");
            }
            Console.WriteLine(" = {0}", giaiThua);
        }
    }
}
